### ok | atribuição simples, inteiro e real
{ int a; float b; float c; a = 1; b = 2.5; c = .5; }

### ok | expressão com precedência e parênteses
{ int a; int b; int c; int d; int x1; a = 1; b = 2; c = 3; d = 4; x1 = (a + b) * c / 2 - d; }

### ok | todos os comparadores
{ int a; int b; int r; a = 1; b = 2; se (a == b) { r = 1; } se (a != b) { r = 2; } se (a < b) { r = 3; } se (a > b) { r = 4; } se (a <= b) { r = 5; } se (a >= b) { r = 6; } }

### ok | se-senao
{ int a; int b; a = 1; se (a > 0) { b = 1; } senao { b = 2; } }

### ok | se aninhado (senão pertence ao se de fora, pelas chaves)
{ int a; int b; int c; a = 1; b = 1; se (a > 0) { se (b > 0) { c = 1; } } senao { c = 2; } }

### ok | enquanto com pular e terminar
{ int i; i = 0; enquanto (i < 10) { i = i + 1; se (i == 3) { pular; } se (i == 8) { terminar; } } }

### ok | faça-enquanto (do-while)
{ int i; i = 0; faca { i = i + 1; } enquanto (i < 5); }

### ok | equivalente ao "repeat ... until A >= 3" dos colegas de sala
{ int a; a = 0; faca { a = a + 1; } enquanto (a < 3); }

### ok | equivalente ao "for i := 1 to 10" dos colegas
{ int i; int x; i = 1; x = 0; enquanto (i <= 10) { x = x + i; i = i + 1; } }

### ok | laços aninhados com terminar no laco de dentro
{ int soma; int i; soma = 0; i = 1; faca { soma = soma + i; enquanto (i != 0) { terminar; } } enquanto (soma < 100); }

### ok | switch com caso e padrão
{ int op; int r; op = 1; switch (op): { caso 1: { r = 10; terminar; } caso 2: { r = 20; } padrao: { r = 0; } } }

### ok | terminar dentro de switch
{ int op; op = 1; switch (op): { caso 1: { terminar; } } }

### ok | comentários de linha e de bloco
{
  int a; int b;
  // comentario de linha
  a = 1; /* comentario
            de bloco */
  b = a; /***/
}

### ok | identificadores com letras, dígitos e palavras parecidas com reservadas
{ int if1; int fim; int senaoX; int enquanto2; if1 = 1; fim = 2; senaoX = 3; enquanto2 = 4; }

### ok | programa em varias linhas
{
  int contador;
  contador = 0;
  enquanto (contador < 3) {
    contador = contador + 1;
  }
}

### erro | falta ponto e virgula
{ int a; a = 1 }

### erro | atribuição com :=
{ int A; int B; int C; int D; A = B; C := 3 + D * 2; }

### erro | do-while em inglês (do ... while)
{ int A; do { A = 1; terminar; } while (A); }

### erro | enquanto sem parenteses e com faca (sintaxe dos colegas)
{ int A; enquanto A <= 10 faca A = A + 1; }

### erro | for ... to ... do (sintaxe Pascal)
{ int i; int x; for i = 1 to 10 do { x = x + i; } }

### erro | repeat ... until (sintaxe Pascal)
{ int A; repeat A = A + 1; until A >= 3; }

### erro | begin ... end no lugar de chaves
begin a = 1; end

### erro | se sem parênteses
{ int a; int b; se a > 1 { b = 2; } }

### erro | se sem chaves
{ int a; int b; se (a > 1) b = 2; }

### erro | bloco vazio
{ int a; enquanto (a < 1) { } }

### erro | terminar sem ponto e virgula
{ int a; enquanto (a < 2) { terminar } }

### erro | faça-enquanto sem ponto e virgula
{ int a; faca { a = 1; } enquanto (a < 2) }

### erro | faca sem enquanto
{ int a; faca { a = 1; } }

### erro | comparação encadeada (%nonassoc)
{ int a; int b; int c; se (a < b < c) { a = 1; } }

### erro | igual (=) usado como comparação
{ int a; int b; int c; se (a = b) { c = 1; } }

### erro | expressão incompleta
{ int a; int b; a = b +; }

### erro | dois programas no mesmo arquivo
{ int a; int b; a = 1; }
{ b = 2; }

### erro | semântico: pular fora de laco
{ pular; }

### erro | semântico: terminar fora de laco ou switch
{ terminar; }

### erro | semântico: pular dentro de switch (sem laco)
{ int x; switch (x): { caso 1: { pular; } } }

### erro | lexico: caractere invalido
{ int a; a = 1 @ 2; }

### erro | lexico: comentario nao fechado
{ int sem; int fechar; int a; /* sem fechar
  a = 1; }

### erro | lexico: identificador com mais de 32 caracteres
{ int abcdefghijabcdefghijabcdefghijabc; }

### ok | declaração com int e float
{ int i; float media; i = 1; media = 2.5; }

### ok | variável declarada e usada
{ int a; int b; a = 10; b = a + 1; }

### ok | declaração dentro de um bloco
{ int x; x = 0; enquanto (x < 3) { int y; y = x; x = x + 1; } }

### ok | variável usada na própria atribuição
{ int a; a = 0; a = a + 1; }

{ a = 1; }

### erro | variável usada sem declaração
{ int c; c = x * 2; }

{ int a; float a; }
